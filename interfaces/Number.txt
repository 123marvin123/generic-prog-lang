namespace Number
{
    fun range(start: Number::Natural, stop: Number::Natural) -> Seq::Seq {}

    fun sum(seq: Seq::Seq) -> Number::Number {
        description { "sum of elements of p(1)" }
        requires { Object::isModelOf(Seq::item(seq), Number::Number) }
        generic() {
            Seq::fold(
                (a: Number, b: Number) => { Number::add(a, b) },
                0,
                seq
            )
        }
    }

    fun isEven(n: Number::Natural) -> Boolean::Boolean {
        description { "is p(1) even?" }
        requires { Object::isModelOf(n, Number::Natural) }
        generic() {
            Object::isEqual(Number::mod(n, 2), 0)
        }
    }

    fun isOdd(n: Number::Natural) -> Boolean::Boolean {
        description { "is p(1) odd?" }
        requires { Object::isModelOf(n, Number::Natural) }
        generic() {
            Object::isEqual(Number::mod(n, 2), 1)
        }
    }
}
